  equation/.style={baseline={([yshift=-0.5ex]current bounding box.center)}},
  spider/.style={dot, draw, fill=#1, minimum width=0.8em},
}
\begin{tikzpicture}[equation, wire flow=north]
  \node[spider=red!45]   (m) at (0,0) {};
  \node[spider=green!45] (d) at (0,1) {};
  \wires{
    @ = { m.south west, m.south east },
    m = d,
    d = { north west = @, north east = @ },
  }
\end{tikzpicture}
